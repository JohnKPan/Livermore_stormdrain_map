# Livermore stormdrain pipeline dependencies.
#
# The fetch scripts need none of this -- fetch_inlets.py and fetch_dem.py are
# pure stdlib (json/csv/math/urllib), as is
# the core of extract_centerline_latlon.py (pandas only for --parquet). So the
# data can be pulled on a bare interpreter; the stack below is for everything
# that reads it.

# --- data ---------------------------------------------------------------
# pandas:  every tabular join in the pipeline.
# pyarrow: the engine behind read_parquet/to_parquet for
#          segments_points_1m.parquet. Never imported by name, so an
#          import scan will call it unused -- it is not. Removing it breaks
#          extract_centerline_latlon.py --parquet and everything after it.
pandas>=2.2
pyarrow>=17

# numpy: imported directly by 8 of the scripts, not merely pulled in by pandas
# and matplotlib. Declared so a fresh install cannot lose it to a transitive
# dependency change.
numpy>=2.0

# --- geospatial ---------------------------------------------------------
# rasterio: read/sample the 3DEP GeoTIFF tiles (add_elevation, preview_dem,
#           map_street_cells)
# pyproj:   WGS84 <-> UTM 10N <-> Web Mercator throughout
# geopandas: reads an --aoi-file in fetch_usgs_lidar.py, which run_pipeline.sh
#           uses to bound the DEM fetch by city_geojson/<city>.geojson
rasterio>=1.4
pyproj>=3.7
geopandas>=1.0

# --- static figures -----------------------------------------------------
# matplotlib:  every PNG (profiles, cell maps, drain charts, DEM previews)
# contextily:  CartoDB basemap tiles under the matplotlib plan views
matplotlib>=3.10
contextily>=1.6

# --- interactive --------------------------------------------------------
# bokeh:       the linked profile+map pages (plot_street_bokeh.py) and the
#              city-wide picker over them (plot_city_overview.py). Also brings
#              jinja2/tornado/pyyaml.
# xyzservices: CartoDB tile providers. Bokeh happens to depend on it, but both
#              scripts import it directly, so it is named here in its own right.
bokeh>=3.9
xyzservices>=2024.9

# plotly: only plot_points_map.py, the city-wide point overview. Optional --
#         drop this line if you do not need that script.
plotly>=6.0
